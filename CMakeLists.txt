cmake_minimum_required(VERSION 3.20)
project(JByteVM VERSION 0.8.0 LANGUAGES C)
set(JBV_PROFILE "full" CACHE STRING "Runtime profile: full, text, core")
set_property(CACHE JBV_PROFILE PROPERTY STRINGS full text core)
if(NOT JBV_PROFILE MATCHES "^(full|text|core)$")
    message(FATAL_ERROR "Unknown JBV_PROFILE: ${JBV_PROFILE}")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_GENERATOR_PLATFORM MATCHES "^(Win32|x64)$")
    message(FATAL_ERROR "Configure MSVC with -A Win32 (default via build.py) or -A x64.")
endif()

add_library(jbytevm SHARED src/vm.c src/classfile.c src/interpreter.c src/objects.c src/memory.c src/exceptions.c src/numeric.c src/native.c src/monitors.c src/access.c src/services.c src/jar.c)
if(JBV_PROFILE STREQUAL "full")
    target_sources(jbytevm PRIVATE src/library.c src/text.c src/decimal.c src/classpath.c src/dynamic.c src/values.c src/reflection.c src/annotations.c src/trace.c "${CMAKE_BINARY_DIR}/generated_classpath.c")
    target_compile_definitions(jbytevm PRIVATE JBV_PROFILE_FULL=1 JBV_PROFILE_CORE=0)
else()
    target_sources(jbytevm PRIVATE src/classpath_stub.c)
    target_compile_definitions(jbytevm PRIVATE JBV_PROFILE_FULL=0)
    if(JBV_PROFILE STREQUAL "text")
        target_sources(jbytevm PRIVATE src/library.c src/text.c src/decimal.c)
        target_compile_definitions(jbytevm PRIVATE JBV_PROFILE_CORE=0)
    else()
        target_sources(jbytevm PRIVATE src/library_core.c)
        target_compile_definitions(jbytevm PRIVATE JBV_PROFILE_CORE=1)
    endif()
endif()
target_include_directories(jbytevm PUBLIC include)
target_include_directories(jbytevm PRIVATE src)
target_compile_definitions(jbytevm PRIVATE JBV_BUILD_DLL)
set(JBV_GC_STRESS 0 CACHE STRING "1: collect and move objects at almost every safepoint (testing)")
target_compile_definitions(jbytevm PRIVATE JBV_GC_STRESS=${JBV_GC_STRESS})
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    enable_language(ASM_MASM)
    target_sources(jbytevm PRIVATE src/native_x86.asm src/interp_x86.asm src/lzma_x86.asm src/decimal_x86.asm src/numeric_x86.asm src/classfile_x86.asm src/jar_x86.asm src/throw_x86.asm src/vm_x86.asm src/text_x86.asm src/classes_x86.asm src/gc_x86.asm src/monitors_x86.asm)
    target_compile_options(jbytevm PRIVATE "$<$<COMPILE_LANGUAGE:ASM_MASM>:/safeseh;/I${CMAKE_SOURCE_DIR}/src>")
endif()
if(MSVC)
    # Internal x86 calls use registers. Public APIs and host callbacks
    # explicitly retain JBV_CALL (__cdecl).
    target_compile_options(jbytevm PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX;/O1;/Oi-;/Gr;/fp:strict>")
    # Only referenced compiler arithmetic helpers are linked into the image.
    # /NODEFAULTLIB still prevents implicit CRT dependencies and startup.
    target_link_libraries(jbytevm PRIVATE libvcruntime libcmt)
    # The inflater keeps byte arrays on the stack; its own bounds checks replace
    # /GS, whose failure handler would import KERNEL32 from the CRT.
    set_source_files_properties(src/jar.c PROPERTIES COMPILE_OPTIONS "/GS-")
    # Allocator callbacks belong to the embedding application. The DLL has no
    # CRT startup or global constructors. The native bridge needs no OS imports.
    target_link_options(jbytevm PRIVATE /NODEFAULTLIB /NOENTRY /INCREMENTAL:NO /OPT:REF /OPT:ICF /MAP)
else()
    target_compile_options(jbytevm PRIVATE -Wall -Wextra -Werror -Os)
endif()

add_executable(jbytevm-demo examples/host.c)
target_compile_definitions(jbytevm-demo PRIVATE JBV_USE_DLL)
target_link_libraries(jbytevm-demo PRIVATE jbytevm)
if(MSVC)
    set_property(TARGET jbytevm-demo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_compile_options(jbytevm-demo PRIVATE /W4 /WX /O1)
endif()
add_executable(jbytevm-bench examples/bench_host.c)
target_compile_definitions(jbytevm-bench PRIVATE JBV_USE_DLL)
target_link_libraries(jbytevm-bench PRIVATE jbytevm)
set_property(TARGET jbytevm-bench PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_options(jbytevm-bench PRIVATE /W4 /WX /O2)
add_executable(jbytevm-run examples/run_host.c)
target_compile_definitions(jbytevm-run PRIVATE JBV_USE_DLL)
target_link_libraries(jbytevm-run PRIVATE jbytevm)
set_property(TARGET jbytevm-run PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_options(jbytevm-run PRIVATE /W4 /WX /O2)
add_executable(jbytevm-native-demo examples/native_host.c)
target_compile_definitions(jbytevm-native-demo PRIVATE JBV_USE_DLL)
target_link_libraries(jbytevm-native-demo PRIVATE jbytevm)
set_property(TARGET jbytevm-native-demo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_options(jbytevm-native-demo PRIVATE /W4 /WX /O1)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    add_library(jbytevm-typed-fixture SHARED tests/typed_fixture.c)
    target_compile_definitions(jbytevm-typed-fixture PRIVATE JBV_USE_DLL)
    target_link_libraries(jbytevm-typed-fixture PRIVATE jbytevm)
    set_property(TARGET jbytevm-typed-fixture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_compile_options(jbytevm-typed-fixture PRIVATE /W4 /WX /O1 /Gr /fp:strict)
endif()
